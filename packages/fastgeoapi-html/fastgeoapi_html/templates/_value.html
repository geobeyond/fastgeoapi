    {% for name, each in value.items() %}
  • {{ name }}: {{ show(each) }}
    • {% endfor %}
{%- elif value is string -%} {%- if value.startswith(("http://", "https://")) and value.lower().endswith((".jpg", ".jpeg", ".png", ".gif", ".bmp", ".webp")) -%} {%- elif value.startswith(("http://", "https://")) and " " not in value -%} {{ value }} {%- else -%} {{ value | urlize }} {%- endif -%} {%- elif value is iterable -%}
    {% for each in value %}
  • {{ show(each) }}
    • {% endfor %}
